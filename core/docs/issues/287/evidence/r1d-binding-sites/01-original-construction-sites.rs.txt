    fn maximum_all_base_sites_two_indexes_retire_then_maximum_roots_in_same_class() {
        with_source(ROWS as u32, |source, header| {
            let temp = TempDir::new("sites_maximum");
            let authority = ScratchAuthority::new(temp.path(), 1).unwrap();
            let mut session = authority
                .begin_sites([0x51; 32], ROWS, ROWS, source.binding_source_id().unwrap())
                .unwrap();
            let selected = scopes(&session, source);
            let owner = status(&authority, session.selection().token());
            assert_header(temp.path(), &owner, selected.sites());
            assert_eq!(session.profile().unwrap().version, 3);
            let capacity = session.site_capacity(selected.sites()).unwrap();
            assert_eq!(
                (capacity.records(), capacity.encoded_bytes()),
                (ROWS, ROWS * 60)
            );
            insert(&mut session, selected.sites(), header, ROWS as u32, |_| {
                true
            });
            assert_eq!(rows(&owner.path, "sites"), ROWS);
            assert_eq!(rows(&owner.path, "roots"), 0);
            let expected = expected_birth(selected.sites(), header, ROWS as u32, |_| true);
            let members = session.site_close_membership(&expected).unwrap();
            let raw_scope = scope_bytes(selected.sites());
            let expected_members =
                oracle::membership(&expected.encode(), Some(oracle::key(&raw_scope, ROWS + 1)));
            assert_eq!(members.encode(), expected_members);
            let mut after = None;
            let mut ordinal = 0u32;
            loop {
                let page = session
                    .site_parent_page(&members, 1, after, SiteParentPageLimit::default())
                    .unwrap();
                assert!(page.records().len() <= 128);
                assert_eq!(page.maximum(), Some(ROWS as u32 - 1));
                for record in page.records() {
                    assert_eq!(
                        record.encode(),
                        oracle::record(
                            &raw_scope,
                            ROWS + 1 - u64::from(ordinal),
                            1,
                            point_bytes(selected.sites(), header, ordinal)
                        )
                    );
                    let (name, child) = source.binding_at(&record.point()).unwrap();
                    assert_eq!(name.as_str(), format!("n{ordinal:08x}"));
                    assert_eq!(child, Some(record.key().serial()));
                    ordinal += 1;
                }
                assert_eq!(
                    page.encode_header(),
                    oracle::parent_page(
                        &expected_members,
                        1,
                        Some(ordinal - 1),
                        Some(ROWS as u32 - 1),
                        page.records().len() as u16,
                        page.eof()
                    )
                );
                after = page.last();
                if page.eof() {
                    break;
                }
            }
            assert_eq!(u64::from(ordinal), ROWS);
            let terminal = session
                .site_parent_page(
                    &members,
                    1,
                    after,
                    SiteParentPageLimit::new(128, 189).unwrap(),
                )
                .unwrap();
            assert!(terminal.eof() && terminal.records().is_empty());
            let mut observations = Vec::with_capacity(128);
            for serial in 2..=ROWS + 1 {
                observations.push(SiteObservation::new(
                    SiteKey::new(selected.sites(), serial).unwrap(),
                    serial % 2 == 0,
                ));
                if observations.len() == 128 {
                    session
                        .site_observe_base_batch(&members, &observations)
                        .unwrap();
                    observations.clear();
                }
            }
            assert!(observations.is_empty());
            let seal = session.site_final_seal(&members).unwrap();
            let mut expected_final = oracle::Transcript::new(raw_scope, true);
            for serial in 2..=ROWS + 1 {
                expected_final.append(
                    serial,
                    if serial % 2 == 0 { 7 } else { 3 },
                    point_bytes(selected.sites(), header, (ROWS + 1 - serial) as u32),
                );
            }
            assert_eq!(seal.encode(), expected_final.seal());
            let mut after = None;
            let mut serial = 2;
            loop {
                let page = session
                    .site_sealed_page(&seal, after, SitePageLimit::default())
                    .unwrap();
                for record in page.records() {
                    assert_eq!(
                        record.encode(),
                        oracle::record(
                            &raw_scope,
                            serial,
                            if serial % 2 == 0 { 7 } else { 3 },
                            point_bytes(selected.sites(), header, (ROWS + 1 - serial) as u32)
                        )
                    );
                    serial += 1;
                }
                assert_eq!(
                    page.encode_header(),
                    oracle::page(
                        &expected_final.seal(),
                        Some(oracle::key(&raw_scope, serial - 1)),
                        page.records().len() as u16,
                        page.eof()
                    )
                );
                after = page.last();
                if page.eof() {
                    break;
                }
            }
            assert_eq!(serial, ROWS + 2);
            let terminal = session
                .site_sealed_page(&seal, after, SitePageLimit::new(128, 171).unwrap())
                .unwrap();
            assert!(terminal.eof() && terminal.records().is_empty());
            let connection = external(&owner.path);
            let site_pages = scalar(&connection, "PRAGMA page_count");
            let index_names:Vec<String>=connection.prepare("SELECT name FROM sqlite_schema WHERE type='index' AND tbl_name='binding_sites' ORDER BY name").unwrap().query_map([],|row|row.get(0)).unwrap().map(Result::unwrap).collect();
            assert_eq!(
                index_names,
                vec!["site_birth_order", "site_existing_parent"]
            );
            drop(connection);
            session.site_retire(&seal).unwrap();
            assert_eq!(rows(&owner.path, "sites"), 0);
            assert_eq!(authority.reserved_bytes().unwrap(), CLASS);
            let connection = external(&owner.path);
            let state: (i64, i64, i64) = connection
                .query_row(
                    "SELECT stage,records,remaining FROM site_owner",
                    [],
                    |row| Ok((row.get(0)?, row.get(1)?, row.get(2)?)),
                )
                .unwrap();
            assert_eq!(state, (4, ROWS as i64, 0));
            for sql in ["SELECT 1 FROM binding_sites LIMIT 1","SELECT 1 FROM binding_sites INDEXED BY site_birth_order LIMIT 1","SELECT 1 FROM binding_sites INDEXED BY site_existing_parent WHERE (flags&1)=1 LIMIT 1"] {assert!(connection.query_row(sql,[],|_|Ok(())).optional().unwrap().is_none());}
            let free_pages = scalar(&connection, "PRAGMA freelist_count");
            assert!(free_pages > 0);
            drop(connection);
            let capacity = session.capacity(selected.roots()).unwrap();
            assert_eq!(
                (capacity.records(), capacity.encoded_bytes()),
                (ROWS, ROWS * 63)
            );
            let root_scope = roots_oracle::scope(
                owner.selector,
                owner.token,
                *session.selection().owner_binding().unwrap(),
                2,
            );
            let mut expected_roots = roots_oracle::Transcript::new(root_scope);
            let mut roots = Vec::with_capacity(128);
            for serial in 1..=ROWS {
                expected_roots.append(serial);
                roots.push(root(selected.roots(), serial));
                if roots.len() == 128 {
                    session.append(selected.roots(), &roots).unwrap();
                    roots.clear();
                }
            }
            let root_seal = session.seal(selected.roots()).unwrap();
            assert_eq!(root_seal.encode(), expected_roots.seal());
            assert_eq!(rows(&owner.path, "roots"), ROWS);
            let final_owner = status(&authority, owner.token);
            let final_pages = scalar(&external(&owner.path), "PRAGMA page_count");
            assert_eq!(final_owner.file, owner.file);
            assert_eq!(final_owner.allocated_bytes, Some(CLASS));
            assert!(site_pages <= 4096 && final_pages <= 4096);
            eprintln!("LFCS3 bounded correctness/resource diagnostic: all-base sites={ROWS}/{}frameB, both indexes={index_names:?}, site_pages={site_pages}, retired_free_pages={free_pages}, roots={ROWS}/{}frameB, final_pages={final_pages}, same_file={:?}, allocation={:?}, retained_class={CLASS}; SiteRecord native_element={}B, returned_window128={}B; caller fixture/names/oracles are excluded from these disk/count claims; no native heap/RSS/global/protected-progress/speed claim",ROWS*60,ROWS*63,owner.file,final_owner.allocated_bytes,std::mem::size_of::<SiteRecord>(),128*std::mem::size_of::<SiteRecord>());

    #[test]
    fn cumulative_declared_class_refuses_before_sql_under_actual_exclusive_lock() {
        with_source(3, |source, header| {
            let temp = TempDir::new("sites_cumulative_admission");
            let authority = ScratchAuthority::new(temp.path(), 1).unwrap();
            let mut session = authority
                .begin_sites([0x57; 32], 1, 2, source.binding_source_id().unwrap())
                .unwrap();
            let selected = scopes(&session, source);
            let owner = status(&authority, session.selection().token());
            insert(&mut session, selected.sites(), header, 2, |_| true);
            let before = std::fs::metadata(&owner.path).unwrap();
            let holder = external(&owner.path);
            let blocked = external(&owner.path);
            holder.execute_batch("BEGIN EXCLUSIVE").unwrap();
            assert_eq!(scalar(&holder, "SELECT records FROM site_owner"), 2);
            let refusal = blocked
                .query_row::<i64, _, _>("SELECT records FROM site_owner", [], |row| row.get(0))
                .unwrap_err();
            assert!(
                matches!(refusal, rusqlite::Error::SqliteFailure(code, _) if code.code == rusqlite::ErrorCode::DatabaseBusy),
                "actual negative control establishes the EXCLUSIVE barrier"
            );
            let error = session
                .site_insert_batch(
                    selected.sites(),
                    &[birth(selected.sites(), header, 2, true)],
                )
                .unwrap_err();
            assert!(
                matches!(
                    error,
                    StorageError::CapacityExceeded {
                        what: "construction scratch site rows",
                        limit: 2,
                        actual: 3
                    }
                ),
                "capacity precedes native reservation/SQL/Busy: {error:?}"
            );
            let current = status(&authority, owner.token);
            assert_eq!(current.disposition, ScratchDisposition::Failed);
            assert_eq!(current.allocated_bytes, Some(CLASS));
            assert!(!current.failure.as_ref().unwrap().contains("site attempt"));
            assert!(!current.quarantined && !current.release_attempted);
            assert_eq!(scalar(&holder, "SELECT records FROM site_owner"), 2);
            assert_eq!(scalar(&holder, "SELECT COUNT(*) FROM binding_sites"), 2);
            assert_eq!(scalar(&holder, "SELECT COUNT(*) FROM directory_roots"), 0);
            let after = std::fs::metadata(&owner.path).unwrap();
            assert_eq!(identity(&before), identity(&after));
            assert_eq!(before.len(), after.len());
            use std::os::unix::fs::MetadataExt;
            assert_eq!(before.blocks(), after.blocks());
            holder.execute_batch("ROLLBACK").unwrap();
            drop(blocked);
            drop(holder);
            session.release().unwrap();
        });
    }

    #[test]
    fn final_seal_rechecks_immutable_base_point_and_redundant_coordinates() {
        for corruption in ["base", "point", "parent", "source"] {
            with_source(2, |source, header| {
                let temp = TempDir::new("sites_immutable_corrupt");
                let authority = ScratchAuthority::new(temp.path(), 1).unwrap();
                let mut session = authority
                    .begin_sites([0x5b; 32], 1, 2, source.binding_source_id().unwrap())
                    .unwrap();
                let selected = scopes(&session, source);
                let owner = status(&authority, session.selection().token());
                insert(&mut session, selected.sites(), header, 2, |_| true);
                let members = session
                    .site_close_membership(&expected_birth(selected.sites(), header, 2, |_| true))
                    .unwrap();
                let connection = external(&owner.path);
                let key = birth(selected.sites(), header, 0, true).key();
                match corruption {
                    "base" => {
                        assert_eq!(
                            connection
                                .execute(
                                    "UPDATE binding_sites SET flags=0 WHERE key=?1",
                                    [key.as_bytes().as_slice()]
                                )
                                .unwrap(),
                            1
                        );
                    }
                    "parent" => {
                        assert_eq!(
                            connection
                                .execute(
                                    "UPDATE binding_sites SET parent=2 WHERE key=?1",
                                    [key.as_bytes().as_slice()]
                                )
                                .unwrap(),
                            1
                        );
                    }
                    _ => {
                        let mut point = point_bytes(selected.sites(), header, 0);
                        point[if corruption == "source" { 0 } else { 16 }] ^= 1;
                        assert_eq!(
                            connection
                                .execute(
                                    "UPDATE binding_sites SET point=?1 WHERE key=?2",
                                    rusqlite::params![point.as_slice(), key.as_bytes().as_slice()]
                                )
                                .unwrap(),
                            1
                        );
                    }
                }
                drop(connection);
                assert!(session.site_final_seal(&members).is_err());
                assert_eq!(rows(&owner.path, "roots"), 0);
                assert!(session.capacity(selected.roots()).is_err());
                assert!(!session.is_quarantined());
                session.release().unwrap();
            });
        }
    }

                .unwrap();
            let seal = session.site_final_seal(&members).unwrap();
            let connection = external(&owner.path);
            connection.execute_batch("CREATE TABLE external_pin(key BLOB PRIMARY KEY REFERENCES binding_sites(key)) STRICT, WITHOUT ROWID").unwrap();
            connection
                .execute(
                    "INSERT INTO external_pin VALUES(?1)",
                    [SiteKey::new(selected.sites(), 130)

                        .as_slice()],
                )
                .unwrap();

                StorageError::Engine(rusqlite::Error::SqliteFailure(..))

            let progress: (i64, i64, Vec<u8>) = external(&owner.path)

            assert_eq!(progress.2, oracle::key(&scope_bytes(selected.sites()), 129));

            assert!(session.capacity(selected.roots()).is_err());
            assert_eq!(rows(&owner.path, "roots"), 0);

            assert_eq!(authority.reserved_bytes().unwrap(), 0);
        });

                    .prepare(&format!("EXPLAIN QUERY PLAN {sql}"))

                    .collect();

            }
